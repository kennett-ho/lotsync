# LotSync runtime dependencies -- backend (reconciliation engine + API).
#
# Unpinned, matching the install commands already documented in
# README.md / api/README.md (`pip install fastapi uvicorn ...`). This
# file exists so `tools/launch.ps1` / `tools/doctor.ps1` / `tools/update.ps1`
# have one place to install from instead of hardcoding a package list --
# it does not change what the project depends on, only where that list
# is written down.
#
# pandas / openpyxl are required by config/settings.py (oms_config.xlsx)
# and every importer under importers/ -- they were not previously listed
# anywhere and were missing from .venv as of Phase 3 Sprint 4; see
# tools/README.md for how this was found.
fastapi
uvicorn
python-multipart
pandas
openpyxl

# Used by tests/test_api_*.py via FastAPI's TestClient (see api/README.md).
httpx

# Used by reports/work_order.py (GET /tasks/work-order) to render the
# printable daily work order as a PDF. Pulls in pillow as its own
# transitive dependency (image handling reportlab's core doesn't need
# for this project's plain-text/vector-only document, but the package
# always requires it).
reportlab

# Sprint 03 (DealerDOH) -- PostgreSQL driver + connection pool for the
# DATABASE_ENGINE=postgres path (DealerDOH DEV runs against Supabase
# PostgreSQL; production keeps SQLite and never exercises these at
# runtime, but they install everywhere so one requirements file serves
# both engines and the dual-engine test suite).
psycopg[binary]
psycopg-pool

# Test-only -- tests/test_work_order.py reads the PDFs
# reports/work_order.py generates back apart (page count, extracted
# text) to assert on real content instead of just "it returned bytes."
# Same "a real, narrow exception to tests/README.md's no-external-
# dependencies note" category as httpx above, not a runtime dependency
# of the app itself.
pypdf

# Sprint 05 (DealerDOH) -- Supabase Auth token verification in
# api/auth.py: PyJWT verifies the ES256-signed access tokens the
# dealerdoh-dev Supabase project issues (against its public JWKS);
# cryptography supplies the ECC P-256 primitives PyJWT needs for that.
# Inert under the default AUTH_MODE=disabled (the production path).
PyJWT
cryptography

# Sprint 11 (DealerDOH, Rails F+G) -- backend error monitoring.
# Initialized ONLY when SENTRY_DSN is set (api/observability.py);
# without it every capture is a silent no-op, so local/CI runs never
# contact Sentry. The [fastapi] extra brings the Starlette/FastAPI
# integration used for request context.
sentry-sdk[fastapi]
